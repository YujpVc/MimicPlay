import numpy as np

def is_success(context):
    """
    自定义任务成功判定逻辑
    任务目标:
    1. Book -> Wooden Shelf
    2. Juice -> White Storage Box
    3. Wine Bottle -> Wooden Tray
    """
    # context.movable_objects 列表顺序根据 env_genesis.py 定义:
    # [0]: book, [1]: juice, [2]: wine_bottle
    
    # 1. 获取各物体当前位置 (get_state返回(pos, quat, vel, ang)，取pos的前两维 x, y)
    book_pos = context.movable_objects[0].get_state()[0][:2]
    juice_pos = context.movable_objects[1].get_state()[0][:2]
    wine_pos = context.movable_objects[2].get_state()[0][:2]
    
    # 2. 定义目标容器的 XY 坐标 (参考 env_genesis.py 中的场景定义)
    # Wooden Shelf: pos=(0.6, 0.65, 0.88) -> Book 的目标
    shelf_target = np.array([0.7, 0.65])
    
    # White Storage Box: pos=(0.35, 0.8, 0.88) -> Juice 的目标
    box_target = np.array([0.35, 0.8])
    
    # Wooden Tray: pos=(0.4, 1.3, 0.88) -> Wine Bottle 的目标
    tray_target = np.array([0.4, 1.3])
    
    # 3. 距离阈值 (单位: 米)，例如 15cm，可根据物体大小适当放宽
    threshold = 0.15
    
    # 4. 计算欧氏距离
    dist_book = np.linalg.norm(book_pos - shelf_target)
    dist_juice = np.linalg.norm(juice_pos - box_target)
    dist_wine = np.linalg.norm(wine_pos - tray_target)
    
    # 5. 判定各子任务是否成功
    book_done = dist_book < threshold
    juice_done = dist_juice < threshold
    wine_done = dist_wine < threshold
    
    # 所有任务都完成才算总成功
    all_done = book_done and juice_done and wine_done
    
    # 返回字典，task 为必须字段，其他为辅助调试信息
    return {
        "task": bool(all_done),
        "sub_book_on_shelf": bool(book_done),
        "sub_juice_in_box": bool(juice_done),
        "sub_wine_on_tray": bool(wine_done),
        "dist_book": float(dist_book),
        "dist_juice": float(dist_juice),
        "dist_wine": float(dist_wine)
    }
